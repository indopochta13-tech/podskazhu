# Шаблон блока Caddy для SoulVoice.
# Вставьте в /etc/caddy/Caddyfile на сервере 201.51.3.63, затем:
#   caddy validate --config /etc/caddy/Caddyfile && systemctl reload caddy
#
# Полная инструкция: voicecapture/DOMAIN_MIGRATION.md
#
# soulvoicee.ru       — сайт о приложении (файлы из /opt/site)
# soulvoicee.ru/app/  — само приложение (Node на 8790)
# soulvoicee.ru/download/, /privacy.html — отдаёт приложение: там лежат APK и документ
# PWA-ассеты (app.js, styles.css, …) — Node fallback, если файла нет в /opt/site
#
# vc.201-51-3-63.sslip.io — портфолио разработчика (/opt/portfolio)

(общее) {
	encode gzip {
		match {
			header Content-Type text/*
			header Content-Type application/json*
			header Content-Type application/javascript*
			header Content-Type application/manifest+json*
			header Content-Type image/svg+xml*
		}
	}
	header {
		Strict-Transport-Security "max-age=31536000"
		X-Content-Type-Options nosniff
		Referrer-Policy no-referrer
		-Server
	}
	log {
		output file /var/log/caddy/access.log {
			roll_size 10MiB
			roll_keep 5
		}
	}
}

soulvoicee.ru, www.soulvoicee.ru {
	redir /privacy /privacy.html permanent
	import общее

	# Приложение и всё, что ему нужно, — на Node.
	handle /app/* {
		uri strip_prefix /app
		reverse_proxy 127.0.0.1:8790
	}
	handle /api/* {
		reverse_proxy 127.0.0.1:8790
	}
	handle /download/* {
		reverse_proxy 127.0.0.1:8790
	}
	handle /app-version.json {
		reverse_proxy 127.0.0.1:8790
	}

	# Лендинг и юр. страницы — /opt/site (корень / явно, иначе Caddy file{} не находит index.html).
	handle / {
		root * /opt/site
		rewrite * /index.html
		file_server
	}
	handle /privacy.html {
		root * /opt/site
		file_server
	}
	handle /offer.html {
		root * /opt/site
		file_server
	}
	handle /shots/* {
		root * /opt/site
		file_server
	}
	# PWA-ассеты и прочее — Node.
	handle {
		reverse_proxy 127.0.0.1:8790
	}
}

vc.201-51-3-63.sslip.io {
	import общее
	redir /portfolio / permanent
	redir /portfolio/ / permanent
	root * /opt/portfolio
	file_server
}
